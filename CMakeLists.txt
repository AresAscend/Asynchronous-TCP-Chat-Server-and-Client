cmake_minimum_required(VERSION 3.20)
project("Asynchronous TCP Chat Server and Client" VERSION 2.0.0 LANGUAGES CXX)

if(POLICY CMP0167)
    # Preserve the FindBoost fallback for header-only/tarball installations
    # while avoiding CMake 3.30's unset-policy warning.
    cmake_policy(SET CMP0167 OLD)
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(ENABLE_ASAN "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(ENABLE_TSAN "Enable ThreadSanitizer" OFF)
if(ENABLE_ASAN AND ENABLE_TSAN)
    message(FATAL_ERROR "ENABLE_ASAN and ENABLE_TSAN cannot both be enabled")
endif()
if((ENABLE_ASAN OR ENABLE_TSAN) AND MSVC)
    message(FATAL_ERROR "The configured sanitizers require Clang or GCC")
endif()

find_package(Boost 1.76 QUIET CONFIG)
if(NOT Boost_FOUND)
    find_package(Boost 1.76 REQUIRED)
endif()
find_package(Threads REQUIRED)

add_library(project_options INTERFACE)
target_include_directories(project_options INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(project_options INTERFACE BOOST_ERROR_CODE_HEADER_ONLY)
if(MSVC)
    target_compile_options(project_options INTERFACE /W4 /permissive-)
else()
    target_compile_options(project_options INTERFACE -Wall -Wextra -Wpedantic)
    if(ENABLE_ASAN)
        target_compile_options(project_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(project_options INTERFACE -fsanitize=address,undefined)
    elseif(ENABLE_TSAN)
        target_compile_options(project_options INTERFACE -fsanitize=thread -fno-omit-frame-pointer)
        target_link_options(project_options INTERFACE -fsanitize=thread)
    endif()
endif()
if(TARGET Boost::headers)
    set(BOOST_HEADERS_TARGET Boost::headers)
elseif(TARGET Boost::boost)
    set(BOOST_HEADERS_TARGET Boost::boost)
else()
    add_library(boost_headers INTERFACE)
    target_include_directories(boost_headers INTERFACE ${Boost_INCLUDE_DIRS})
    set(BOOST_HEADERS_TARGET boost_headers)
endif()
target_link_libraries(project_options INTERFACE ${BOOST_HEADERS_TARGET} Threads::Threads)

add_executable(chatApp chatRoom.cpp)
target_link_libraries(chatApp PRIVATE project_options)
if(UNIX)
    add_executable(clientApp client.cpp)
    target_link_libraries(clientApp PRIVATE project_options)
endif()

install(TARGETS chatApp RUNTIME DESTINATION bin)
if(UNIX)
    install(TARGETS clientApp RUNTIME DESTINATION bin)
endif()

include(CTest)
if(BUILD_TESTING)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_executable(message_tests tests/message_tests.cpp)
    target_link_libraries(message_tests PRIVATE project_options)
    add_test(NAME message_tests COMMAND message_tests)
    add_test(NAME integration_test COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration_test.py --server $<TARGET_FILE:chatApp>)
    add_test(NAME slow_peer_test COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/slow_peer_test.py --server $<TARGET_FILE:chatApp>)
    if(UNIX)
        add_test(NAME client_pipe_test COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/client_pipe_test.py --server $<TARGET_FILE:chatApp> --client $<TARGET_FILE:clientApp>)
    endif()
endif()
